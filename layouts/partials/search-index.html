{{- /* Builds the global search index from site content: every play, every spell, and the reference pages with their section headings. (.Params.threat_tags | default slice) " ") .Params.table_phrase -}} {{- $items = $items | append (dict "title" (printf "%s: %s" .Params.play_id .Title) "category" $category "summary" (.Params.summary | markdownify | plainify | htmlUnescape) "keywords" ($keywords | markdownify | plainify | htmlUnescape) "url" (printf "%s#%s" $playsList.RelPermalink (lower .Params.play_id)) ) -}} {{- end -}} {{- /* Spells: link to the card anchor on the spellbook page */ -}} {{- $spellbook := site.GetPage "/spellbook" -}} {{- range sort (where site.RegularPages "Section" "spellbook") "Params.level" -}} {{- $flags | append "Concentration" }}{{ end -}} {{- if .Params.ritual }}{{ $flags = $flags | append "Ritual" }}{{ end -}} {{- if .Params.sculpt_spells }}{{ $flags = $flags | append "Sculpt Spells" }}{{ end -}} {{- if .Params.potent_cantrip }}{{ $flags = $flags | append "Potent Cantrip" }}{{ end -}} {{- if .Params.prepared }}{{ $flags = $flags | append "Prepared" }}{{ end -}} {{- $items = $items | append (dict "title" .Title "category" $category "summary" (printf "%s · %s · %s · %s" .Params.school .Params.casting_time .Params.range .Params.duration) "keywords" (printf "%s %s %s" (delimit $flags " ") .Params.upcast_notes (.Content | plainify | htmlUnescape)) "url" (printf "%s#%s" $spellbook.RelPermalink (.Title | lower | urlize)) ) -}} {{- end -}} {{- /* Reference pages and their second-level {{- end -}} {{- $js := printf "window.JOE_SEARCH_INDEX = %s;" (jsonify $items) | resources.FromString "js/search-index.js" | minify | fingerprint -}} {{- return $js -}}
